## vm1-dns
vks@vm1-dns:~$ # BEFORE the rule: any LAN host reaches the backends directly
vks@vm1-dns:~$ for u in 13:3001 14:3002; do curl -sS --connect-timeout 3 -o /dev/null -w "192.168.64.$u -> HTTP %{ht
tp_code}\n" http://192.168.64.$u/api/status; done

tp_code}\n" http://192.168.64.$u/api/status; done
